------------------------------------------------------------ -- DISTANCE ------------------------------------------------------------ local function distance( x1,y1, x2,y2 ) local dx = x2-x1 local dy = y2-y1 return math.sqrt( dx*dx+dy*dy ) end ------------------------------------------------------------ -- RECT COLLISION ------------------------------------------------------------ local function rectCollision( x1,y1,w1,h1, x2,y2,w2,h2 ) return x1 < x2+w2 and x1+w1 > x2 and y1 < y2+h2 and y1+h1 > y2 end ------------------------------------------------------------ -- CAR BOX ------------------------------------------------------------ local function carBox() return car.x-car.width/2, car.y-car.height/2, car.width, car.height end ------------------------------------------------------------ -- COLLISION ------------------------------------------------------------ local function checkCollision() local x,y,w,h = carBox() -------------------------------------------------------- -- WALLS -------------------------------------------------------- for _,wall in ipairs(walls) do if rectCollision( x,y,w,h, wall.x, wall.y, wall.w, wall.h ) then return true end end -------------------------------------------------------- -- PARKED CARS -------------------------------------------------------- for _,other in ipairs(parkedCars) do if rectCollision( x,y,w,h, other.x-other.width/2, other.y-other.height/2, other.width, other.height ) then return true end end -------------------------------------------------------- -- OBSTACLES -------------------------------------------------------- for _,ob in ipairs(obstacles) do if rectCollision( x,y,w,h, ob.x, ob.y, ob.w, ob.h ) then return true end end return false end ------------------------------------------------------------ -- PARK CHECK ------------------------------------------------------------ local function checkParking() local px = parking.x+ parking.width/2 local py = parking.y+ parking.height/2 local d = distance( car.x, car.y, px, py ) return d < 45 and math.abs(car.speed) < 20 end
